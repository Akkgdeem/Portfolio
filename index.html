Excel VBA Macro: HTML Cleaner

Sub RemoveHTML_AllSheets_Safe()
    Dim ws As Worksheet, cell As Range, regEx As Object
    Set regEx = CreateObject("VBScript.RegExp")
    regEx.Pattern = "<[^>]+>"
    Application.ScreenUpdating = False
    For Each ws In ThisWorkbook.Worksheets
        For Each cell In ws.UsedRange.Cells
            If Not IsError(cell.Value) And VarType(cell.Value) = vbString Then
                cell.Value = Application.WorksheetFunction.Trim(regEx.Replace(cell.Value, ""))
            End If
        Next
    Next
    Application.ScreenUpdating = True
    MsgBox "HTML removed from all sheets."
End Sub
